function getSignedArea(contour) {
    let sum = 0;
    for (let i = 0; i < contour.length; i++) {
        const p1 = contour[i].anchor;
        const p2 = contour[(i + 1) % contour.length].anchor;
        sum += (p2.x - p1.x) * (p2.y + p1.y);
    }
    return sum;
}
